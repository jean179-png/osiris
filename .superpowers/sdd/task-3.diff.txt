+import { NextRequest, NextResponse } from 'next/server';
+import { parseBbox, bboxKey } from '@/lib/layers/bbox';
+import { cachedFc } from '@/lib/layers/cache';
+import { emptyFc } from '@/lib/layers/geojson';
+import { mapNoaaWrecksToFc } from '@/lib/layers/wrecks';
+
+export async function GET(req: NextRequest) {
+  const bbox = parseBbox(req.nextUrl.searchParams.get('bbox'));
+  if (!bbox) return NextResponse.json(emptyFc());
+  const key = bboxKey('wrecks', bbox);
+  const fc = await cachedFc(key, 5 * 60_000, async () => {
+    const geometry = `${bbox.west},${bbox.south},${bbox.east},${bbox.north}`;
+    const url =
+      'https://gis.charttools.noaa.gov/arcgis/rest/services/public_wrecks/Wrecks_And_Obstructions/MapServer/0/query?' +
+      new URLSearchParams({
+        where: '1=1',
+        geometry,
+        geometryType: 'esriGeometryEnvelope',
+        inSR: '4326',
+        spatialRel: 'esriSpatialRelIntersects',
+        outFields: '*',
+        returnGeometry: 'true',
+        f: 'geojson',
+        resultRecordCount: '2000',
+      }).toString();
+    try {
+      const res = await fetch(url, {
+        signal: AbortSignal.timeout(12_000),
+        headers: { 'User-Agent': 'osiris-layers/1.0', Accept: 'application/json' },
+      });
+      if (!res.ok) return emptyFc();
+      const ct = res.headers.get('content-type') ?? '';
+      if (!ct.includes('json') && !ct.includes('geojson')) {
+        // NOAA sometimes returns HTML error pages with 200
+        const text = await res.text();
+        try {
+          return mapNoaaWrecksToFc(JSON.parse(text));
+        } catch {
+          return emptyFc();
+        }
+      }
+      return mapNoaaWrecksToFc(await res.json());
+    } catch {
+      return emptyFc();
+    }
+  });
+  return NextResponse.json(fc, { headers: { 'Cache-Control': 'public, s-maxage=60' } });
+}

+import { describe, it, expect } from 'vitest';
+import { mapNoaaWrecksToFc } from './wrecks';
+
+describe('mapNoaaWrecksToFc', () => {
+  it('maps Point features to ground_wrecks', () => {
+    const fc = mapNoaaWrecksToFc({
+      type: 'FeatureCollection',
+      features: [
+        {
+          type: 'Feature',
+          properties: {
+            OBJECTID: 42,
+            vesslterms: 'Wreck',
+            history: 'SS Example',
+            depth: 12,
+            chart: '13270',
+            recrd: 1001,
+          },
+          geometry: { type: 'Point', coordinates: [-70.95, 42.35] },
+        },
+      ],
+    });
+    expect(fc.features).toHaveLength(1);
+    const p = fc.features[0].properties;
+    expect(p?.kind).toBe('ground_wrecks');
+    expect(p?.name).toBe('SS Example');
+    expect(p?.featureType).toBe('Wreck');
+    expect(p?.attribution).toBe('NOAA');
+    expect(p?.depth).toBe(12);
+    expect(fc.features[0].geometry).toEqual({
+      type: 'Point',
+      coordinates: [-70.95, 42.35],
+    });
+  });
+
+  it('expands MultiPoint into one feature per coordinate', () => {
+    const fc = mapNoaaWrecksToFc({
+      type: 'FeatureCollection',
+      features: [
+        {
+          type: 'Feature',
+          properties: { OBJECTID: 7, VESSLTERMS: 'Obstruction' },
+          geometry: {
+            type: 'MultiPoint',
+            coordinates: [
+              [-71.0, 42.3],
+              [-71.01, 42.31],
+            ],
+          },
+        },
+      ],
+    });
+    expect(fc.features).toHaveLength(2);
+    expect(fc.features[0].properties?.featureType).toBe('Obstruction');
+    expect(fc.features[0].properties?.kind).toBe('ground_wrecks');
+    expect(fc.features[1].geometry).toEqual({
+      type: 'Point',
+      coordinates: [-71.01, 42.31],
+    });
+  });
+
+  it('returns empty for missing features', () => {
+    expect(mapNoaaWrecksToFc({}).features).toHaveLength(0);
+    expect(mapNoaaWrecksToFc(null).features).toHaveLength(0);
+  });
+
+  it('skips non-point geometries and bad coordinates', () => {
+    const fc = mapNoaaWrecksToFc({

+      ],

+      type: 'FeatureCollection',

+    });

+import type { Feature, FeatureCollection } from 'geojson';
+import { pointFeature, emptyFc, capFeatures } from './geojson';
+
+function prop(
+  props: Record<string, unknown>,
+  ...keys: string[]
+): unknown {
+  for (const k of keys) {
+    if (props[k] != null && props[k] !== '') return props[k];
+  }
+  return undefined;
+}
+
+function mapProps(raw: Record<string, unknown>): Record<string, unknown> {
+  const featureType =
+    prop(raw, 'featureType', 'FeatureType', 'vesslterms', 'VESSLTERMS') ?? null;
+  const name =
+    prop(raw, 'name', 'NAME', 'history', 'HISTORY', 'vesslterms', 'VESSLTERMS') ??
+    'Wreck';
+  return {
+    kind: 'ground_wrecks',
+    name: typeof name === 'string' ? name : String(name),
+    featureType: featureType == null ? null : String(featureType),
+    depth: prop(raw, 'depth', 'DEPTH') ?? null,
+    chart: prop(raw, 'chart', 'CHART') ?? null,
+    record: prop(raw, 'recrd', 'RECRD', 'record') ?? null,
+    attribution: 'NOAA',
+  };
+}
+
+/** NOAA ArcGIS GeoJSON is loosely shaped; keep the mapper permissive. */
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+export function mapNoaaWrecksToFc(data: any): FeatureCollection {
+  const incoming = data?.features;
+  if (!Array.isArray(incoming)) return emptyFc();
+  const features: Feature[] = [];
+  for (const f of incoming) {
+    const geom = f?.geometry;
+    const raw = (f?.properties ?? {}) as Record<string, unknown>;
+    const mapped = mapProps(raw);
+    const idBase =
+      prop(raw, 'OBJECTID', 'objectid', 'OBJECTID_1', 'recrd', 'RECRD') ??
+      features.length;
+
+    if (geom?.type === 'Point' && Array.isArray(geom.coordinates)) {
+      const [lng, lat] = geom.coordinates;
+      if (Number.isFinite(lng) && Number.isFinite(lat)) {
+        features.push(pointFeature(`wreck-${idBase}`, lng, lat, mapped));
+      }
+      continue;
+    }
+
+    if (geom?.type === 'MultiPoint' && Array.isArray(geom.coordinates)) {
+      for (let i = 0; i < geom.coordinates.length; i++) {
+        const coord = geom.coordinates[i];
+        if (!Array.isArray(coord)) continue;
+        const [lng, lat] = coord;
+        if (!Number.isFinite(lng) || !Number.isFinite(lat)) continue;
+        features.push(pointFeature(`wreck-${idBase}-${i}`, lng, lat, mapped));
+      }
+    }
+  }
+  return capFeatures({ type: 'FeatureCollection', features }, 2000);
+}
