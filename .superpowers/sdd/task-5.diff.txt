+import { NextRequest, NextResponse } from 'next/server';
+import { parseBbox, bboxKey } from '@/lib/layers/bbox';

+const TTL_MS = 10 * 60_000;

+export async function GET(req: NextRequest) {
+  const bbox = parseBbox(req.nextUrl.searchParams.get('bbox'));
+  if (!bbox) return NextResponse.json(emptyFc());
+
+  const zRaw = req.nextUrl.searchParams.get('z');
+  const z = zRaw != null ? Number(zRaw) : NaN;
+  if (!Number.isFinite(z) || z < MIN_ZOOM) {
+    return NextResponse.json(emptyFc());
+  }
+
+  const key = bboxKey('abandoned', bbox);
+  const fc = await cachedFc(key, TTL_MS, async () => {
+    try {
+      const data = await overpassQuery(abandonedQl(bbox));
+      return mapAbandonedElementsToFc(data);
+    } catch {
+      return emptyFc();
+    }
+  });
+  return NextResponse.json(fc, { headers: { 'Cache-Control': 'public, s-maxage=60' } });

+import type { Feature, FeatureCollection, LineString, Position } from 'geojson';
+import type { BBox } from './types';
+import { fillBbox } from './overpass';
+import { pointFeature, featureOf, emptyFc, capFeatures } from './geojson';
+
+const ABANDONED_QL = `[out:json][timeout:25];
+(
+  way["railway"~"^(abandoned|disused)$"]({{bbox}});
+  node["railway"~"^(abandoned|disused)$"]({{bbox}});
+  node["abandoned:industrial"]({{bbox}});
+  node["industrial"]["disused"="yes"]({{bbox}});
+);
+out center body geom;
+`;
+
+export function abandonedQl(bbox: BBox): string {
+  return fillBbox(ABANDONED_QL, bbox);
+}
+
+function isRail(tags: Record<string, string>): boolean {
+  const rw = tags.railway;
+  return rw === 'abandoned' || rw === 'disused';
+}
+
+function elementPoint(
+  // eslint-disable-next-line @typescript-eslint/no-explicit-any
+  el: any,
+): { lng: number; lat: number } | null {
+  const lat = el?.lat ?? el?.center?.lat;
+  const lng = el?.lon ?? el?.center?.lon;
+  if (!Number.isFinite(lat) || !Number.isFinite(lng)) return null;
+  return { lng, lat };
+}
+
+function wayLineGeometry(
+  // eslint-disable-next-line @typescript-eslint/no-explicit-any
+  el: any,
+): LineString | null {
+  const geom = el?.geometry;
+  if (!Array.isArray(geom) || geom.length < 2) return null;
+  const coordinates: Position[] = [];
+  for (const g of geom) {
+    const lat = g?.lat;
+    const lon = g?.lon;
+    if (!Number.isFinite(lat) || !Number.isFinite(lon)) continue;
+    coordinates.push([lon, lat]);
+  }
+  if (coordinates.length < 2) return null;
+  return { type: 'LineString', coordinates };
+}
+
+/** Map Overpass JSON elements to abandoned rail/industrial features. */
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+export function mapAbandonedElementsToFc(data: any): FeatureCollection {
+  const elements = data?.elements;
+  if (!Array.isArray(elements)) return emptyFc();
+  const features: Feature[] = [];
+  for (const el of elements) {
+    const tags = (el?.tags ?? {}) as Record<string, string>;
+    const subkind = isRail(tags) ? 'rail' : 'industrial';
+    const id = el?.id ?? features.length;
+    const props = {
+      kind: 'ground_abandoned',
+      subkind,
+      name: tags.name ?? tags['name:en'] ?? null,
+      railway: tags.railway ?? null,
+      industrial: tags.industrial ?? tags['abandoned:industrial'] ?? null,
+      osmId: id,
+      osmType: el?.type ?? 'node',
+      attribution: 'OpenStreetMap',
+    };
+
+    if (el?.type === 'way') {
+      const line = wayLineGeometry(el);
+      if (line) {
+        features.push(featureOf(`abandoned-way-${id}`, line, props));
+        continue;
+      }
+      const pt = elementPoint(el);
+      if (pt) {
+        features.push(pointFeature(`abandoned-way-${id}`, pt.lng, pt.lat, props));
+      }
+      continue;
+    }
+
+    const pt = elementPoint(el);
+    if (!pt) continue;
+    features.push(
+      pointFeature(`abandoned-${el?.type ?? 'n'}-${id}`, pt.lng, pt.lat, props),
+    );
+  }
+  return capFeatures({ type: 'FeatureCollection', features }, 2000);
+}

+const NIGHTLIFE_QL = `[out:json][timeout:25];
+(
+  node["amenity"~"^(bar|pub|nightclub|biergarten)$"]({{bbox}});
+  node["place"="square"]({{bbox}});
+  node["highway"="pedestrian"]["area"="yes"]({{bbox}});
+);
+out center body;
+`;
+
+export function nightlifeQl(bbox: BBox): string {
+  return fillBbox(NIGHTLIFE_QL, bbox);

+
+function elementPoint(
+  // eslint-disable-next-line @typescript-eslint/no-explicit-any
+  el: any,
+): { lng: number; lat: number } | null {

+  if (!Number.isFinite(lat) || !Number.isFinite(lng)) return null;
+  return { lng, lat };

+/** Map Overpass JSON elements to nightlife Point features. */
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+export function mapNightlifeElementsToFc(data: any): FeatureCollection {
+  const elements = data?.elements;
+  if (!Array.isArray(elements)) return emptyFc();
+  const features: Feature[] = [];
+  for (const el of elements) {
+    const pt = elementPoint(el);
+    if (!pt) continue;

+        place: tags.place ?? null,
+        highway: tags.highway ?? null,
+        osmId: id,
+        osmType: el?.type ?? 'node',
+        attribution: 'OpenStreetMap',
+      }),
+    );
+  }

+import type { BBox } from './types';
+
+const ENDPOINTS = [
+  'https://overpass-api.de/api/interpreter',
+  'https://overpass.kumi.systems/api/interpreter',
+] as const;
+
+const TIMEOUT_MS = 25_000;
+const USER_AGENT = 'osiris-layers/1.0';
+
+/** Overpass bbox filter fragment: south,west,north,east */
+export function bboxToOverpass(b: BBox): string {
+  return `${b.south},${b.west},${b.north},${b.east}`;
+}
+
+export function fillBbox(ql: string, b: BBox): string {
+  return ql.replaceAll('{{bbox}}', bboxToOverpass(b));
+}
+
+/**
+ * POST Overpass QL to primary interpreter, then kumi fallback.
+ * Throws if both endpoints fail.
+ */
+// eslint-disable-next-line @typescript-eslint/no-explicit-any
+export async function overpassQuery(ql: string): Promise<any> {
+  let lastError: unknown;
+  for (const url of ENDPOINTS) {
+    try {
+      const res = await fetch(url, {
+        method: 'POST',
+        body: `data=${encodeURIComponent(ql)}`,
+        signal: AbortSignal.timeout(TIMEOUT_MS),
+        headers: {
+          'User-Agent': USER_AGENT,
+          'Content-Type': 'application/x-www-form-urlencoded',
+          Accept: 'application/json',
+        },
+      });
+      if (!res.ok) {
+        lastError = new Error(`Overpass ${url} HTTP ${res.status}`);
+        continue;
+      }
+      return await res.json();
+    } catch (err) {
+      lastError = err;
+    }
+  }
+  throw lastError instanceof Error
+    ? lastError
+    : new Error('Overpass query failed on all endpoints');
+}
